1安装与初始化
Vue CLI 安装（推荐）
# 通过 vue add 安装（自动配置）
vue add vuetify    # 选 Vuetify 2 - Vue CLI (recommended)

# 或手动 npm 安装
npm install vuetify@^2.7.2
main.js 配置
import Vue       from 'vue'
import Vuetify   from 'vuetify/lib'
import 'vuetify/dist/vuetify.min.css'
import '@mdi/font/css/materialdesignicons.css'   // MDI 图标

Vue.use(Vuetify)

const vuetify = new Vuetify({
  theme: {
    themes: {
      light: {
        primary:   '#1976D2',
        secondary: '#424242',
        accent:    '#82B1FF',
        error:     '#FF5252',
        success:   '#4CAF50',
        warning:   '#FFC107',
      }
    }
  }
})

new Vue({ el: '#app', vuetify, render: h => h(App) })
// ⚠️ 必须把 vuetify 实例传入 new Vue，否则所有主题/颜色失效
⚠️ 最重要的规则：所有 Vuetify 组件必须被包裹在 <v-app> 根组件内， 否则主题、RTL、暗色模式、Portal（dialog/snackbar）均无法正常工作。
CDN 引入（本文 Demo 使用方式）

// JS 初始化

new Vue({

2核心：v-app 应用架构
Vuetify 提供了一套完整的应用布局组件，通过语义化组件描述应用结构， Vuetify 内部会自动计算各区域的 z-index、尺寸和安全距离（inset）。 
<!-- 完整后台管理系统布局骨架 -->
<v-app>
  <!-- 顶部导航栏 -->
  <v-app-bar app color="primary" dark>
    <v-app-bar-nav-icon @click="drawer = !drawer" />
    <v-toolbar-title>管理系统</v-toolbar-title>
    <v-spacer />
    <v-btn icon><v-icon>mdi-account</v-icon></v-btn>
  </v-app-bar>

  <!-- 侧边导航 -->
  <v-navigation-drawer v-model="drawer" app>
    <v-list nav dense>
      <v-list-item v-for="item in navItems" :key="item.title" :to="item.path" link>
        <v-list-item-icon><v-icon>{{ item.icon }}</v-icon></v-list-item-icon>
        <v-list-item-title>{{ item.title }}</v-list-item-title>
      </v-list-item>
    </v-list>
  </v-navigation-drawer>

  <!-- 主内容区（⚠️ app prop 告知 Vuetify 自动计算偏移）-->
  <v-main>
    <v-container fluid>
      <router-view />
    </v-container>
  </v-main>

  <!-- 底部栏（可选）-->
  <v-footer app>版权信息</v-footer>
</v-app>
app prop 的作用：带有 app 的布局组件（v-app-bar、v-navigation-drawer、v-footer） 会自动告知 v-main 预留对应的内边距，无需手动计算高度偏移。
3栅格系统（v-container / v-row / v-col）
与 Element UI 相同的 12列栅格（注意 Element UI 是 24列！Vuetify 是 12列）， 断点：xs sm md lg xl，响应式 cols 写法：:sm="6" :md="4"。 
<!-- Vuetify 12列栅格 -->
<v-container>
  <v-row>
    <v-col cols="12" sm="6" md="3">   <-- 手机全宽/平板半宽/桌面1/4
      卡片 1
    </v-col>
  </v-row>

  <!-- 常用辅助 prop -->
  <v-row dense>       <-- dense: 减小列间距（4px → 2px）
  <v-row no-gutters> <-- 去掉所有 gutter
  <v-row justify="center" align="center">  <-- flex 对齐
</v-container>

<!-- ⚠️ 12列 vs 24列 记忆口诀 -->
// Vuetify: cols="6" = 50%    （一半）
// Element: :span="12" = 50%  （一半）

5文本框 v-text-field（及其他输入组件）
Vuetify 的所有输入组件（v-text-field、v-select、v-textarea、v-autocomplete 等） 共享一套基础 Props，统一的 outlined / filled / solo 外观模式。 
<!-- 三种外观模式 -->
<v-text-field label="默认（下划线）"   v-model="val" />
<v-text-field label="outlined（边框）" outlined v-model="val" />
<v-text-field label="filled（填充）"   filled   v-model="val" />
<v-text-field label="solo（独立）"     solo     v-model="val" />

<!-- 常用 Props -->
<v-text-field
  v-model="keyword"
  label="搜索"
  outlined
  dense                       <-- 紧凑高度
  clearable
  prepend-inner-icon="mdi-magnify"
  hide-details                <-- 隐藏错误信息占位（紧凑布局用）
  @keyup.enter="search"
/>

<!-- 密码框 -->
<v-text-field
  v-model="pwd"
  label="密码"
  outlined
  :type="showPwd ? 'text' : 'password'"
  :append-icon="showPwd ? 'mdi-eye' : 'mdi-eye-off'"
  @click:append="showPwd = !showPwd"
/>

<!-- v-select 下拉 -->
<v-select
  v-model="role"
  :items="roleItems"
  item-text="label"     <-- 显示哪个字段
  item-value="value"   <-- 绑定哪个字段
  label="角色"
  outlined
  dense
  clearable
/>
6表单与验证
Vuetify 的表单验证通过 :rules 传入函数数组，每个函数接收当前值， 返回 true 表示通过，返回字符串则作为错误信息展示。与 Element UI 的对象式规则不同。 
<!-- 模板 -->
<v-form ref="form" v-model="formValid">    <-- v-model 绑定整体是否合法
  <v-text-field
    v-model="form.username"
    label="用户名"
    outlined
    :rules="usernameRules"
  />
  <v-btn color="primary" :disabled="!formValid" @click="submit">提交</v-btn>
</v-form>

// JS 验证规则（函数数组）
usernameRules: [
  v => !!v              || '用户名不能为空',
  v => v.length >= 3   || '长度至少 3 位',
  v => v.length <= 20  || '长度不超过 20 位',
  v => /^[a-zA-Z0-9_]+$/.test(v) || '只能包含字母数字下划线'
],
emailRules: [
  v => !!v || '邮箱不能为空',
  v => /.+@.+\..+/.test(v) || '邮箱格式不正确'
]

// 通过 ref 调用的方法
this.$refs.form.validate()       // 触发所有字段校验，返回 Boolean
this.$refs.form.reset()          // 重置值 + 清除校验状态
this.$refs.form.resetValidation() // 仅清除校验状态（不改值）
7数据表格 v-data-table（重点）
v-data-table 是 Vuetify 最强大的组件之一，内置客户端排序、分页、搜索过滤， 不需要手动处理这些逻辑（与 Element UI 的 el-table 需要手动实现不同）。 
<!-- headers 定义列 -->
<v-data-table
  :headers="headers"
  :items="list"
  :search="keyword"       <-- 传入就自动实现全局搜索过滤！
  :loading="loading"
  :items-per-page="10"
  item-key="id"
  show-select            <-- 显示复选框（多选）
  v-model="selected"      <-- 绑定选中行
>
  <!-- 自定义列（作用域插槽 item.列名）-->
  <template v-slot:item.status="{ item }">
    <v-chip :color="item.status ? 'success' : 'error'" small dark>
      {{ item.status ? '启用' : '禁用' }}
    </v-chip>
  </template>

  <!-- 操作列 -->
  <template v-slot:item.actions="{ item }">
    <v-icon small class="mr-2" @click="edit(item)">mdi-pencil</v-icon>
    <v-icon small color="error" @click="del(item)">mdi-delete</v-icon>
  </template>
</v-data-table>

// headers 格式
headers: [
  { text: '姓名',  value: 'name',    sortable: true  },
  { text: '邮箱',  value: 'email',   sortable: false },
  { text: '状态',  value: 'status',  align: 'center' },
  { text: '操作',  value: 'actions', sortable: false, align: 'end' }
]

9对话框 v-dialog
<!-- v-dialog 使用 v-model 控制显示 -->
<v-dialog v-model="dialogVisible" max-width="500px" persistent>
  <!-- persistent: 点遮罩不关闭（类似 Element :close-on-click-modal="false"）-->

  <!-- 触发器（作用域插槽 activator）-->
  <template v-slot:activator="{ on, attrs }">
    <v-btn v-bind="attrs" v-on="on" color="primary">打开弹窗</v-btn>
  </template>

  <!-- 弹窗内容（通常用 v-card 包裹）-->
  <v-card>
    <v-card-title>新建用户</v-card-title>
    <v-card-text>
      <v-text-field v-model="form.name" label="姓名" outlined dense />
    </v-card-text>
    <v-card-actions>
      <v-spacer />
      <v-btn text @click="dialogVisible = false">取消</v-btn>
      <v-btn color="primary" :loading="saving" @click="save">保存</v-btn>
    </v-card-actions>
  </v-card>
</v-dialog>
Vuetify Dialog vs Element Dialog 的区别： Vuetify 的 Dialog 不靠 :visible.sync 而是普通 v-model 控制； 触发按钮可以通过 activator 插槽内嵌（无需额外 data 变量），也可用独立 boolean data 控制。
10提示：v-snackbar / v-alert
<!-- v-snackbar：底部弹出提示（类似 Element $message）-->
<v-snackbar
  v-model="snackbar"
  :timeout="3000"
  :color="snackColor"
  top                   <-- 位置 top/bottom/left/right 组合
>
  {{ snackText }}
  <template v-slot:action>
    <v-btn text @click="snackbar = false">关闭</v-btn>
  </template>
</v-snackbar>

// 封装为全局方法（类似 $message）
methods: {
  toast(text, color = 'success') {
    this.snackText  = text
    this.snackColor = color
    this.snackbar   = true
  }
}
// 调用
this.toast('保存成功')
this.toast('删除失败', 'error')

<!-- v-alert：页面内嵌提示框（类似 Element $message + 常驻版）-->
<v-alert type="success" dense dismissible>操作成功！</v-alert>
<v-alert type="error"   outlined>错误信息</v-alert>
<v-alert type="warning" border="left" colored-border :elevation="2">
  带左边框警告
</v-alert>

14Vuetify vs Element UI 全面对比
维度Vuetify 2.xElement UI 2.x
设计规范Google Material Design阿里 Element 设计规范
适用场景面向 C 端/消费者产品、跨平台外观面向 B 端/后台管理系统（国内主流）
栅格列数12 列（cols="6" = 50%）24 列（:span="12" = 50%）
表单验证函数数组，每个函数返回 true 或错误字符串规则对象数组，含 required/type/min/max 等字段
表格内置功能内置排序+分页+搜索过滤（客户端）排序需手动监听事件，分页独立组件
主题定制new Vuetify({ theme: { themes: { light: {...} } } })修改 SCSS 变量，需自定义主题编译
图标系统内置 MDI / Font Awesome 等多套图标自带 el-icon-* 类名图标（较少）
暗色模式官方原生支持（dark prop / 全局切换）不支持，需第三方或手动实现
消息提示v-snackbar 组件（需手动管理 boolean）this.$message() 直接调用更简单
Dialog 控制普通 v-model:visible.sync（.sync 修饰符）
Bundle 大小完整引入约 ~250KB gzip完整引入约 ~160KB gzip
国内生态文档/中文社区相对较少中文文档完善，国内项目首选
SSR/Nuxt官方 @nuxtjs/vuetify 模块支持良好需手动配置，无官方 Nuxt 模块
选型建议： 国内 B 端后台管理系统首选 Element UI（生态成熟、文档完善、符合国内审美）； C 端/ToC 产品、国际化项目、需要暗色模式/Material Design 风格时选 Vuetify。
15综合实战（Vuetify 版 CRUD 页面）
利用 v-data-table 内置搜索 + 分页，配合 v-dialog 实现新建/编辑弹窗，比 Element UI 代码量更少。
<!-- Vuetify CRUD 页面核心模板 -->
<v-app>
  <v-container>
    <!-- 工具栏 -->
    <v-row class="mb-4" align="center">
      <v-col cols="12" sm="6">
        <v-text-field v-model="search" label="搜索"
          prepend-inner-icon="mdi-magnify" outlined dense hide-details />
      </v-col>
      <v-col cols="auto">
        <v-btn color="primary" @click="openDialog()">
          <v-icon left>mdi-plus</v-icon>新建
        </v-btn>
      </v-col>
    </v-row>

    <!-- 表格（内置搜索 :search="search" 即可，无需额外代码）-->
    <v-data-table
      :headers="headers" :items="list" :search="search"
      :loading="loading" class="elevation-1"
    >
      <template v-slot:item.status="{ item }">
        <v-chip :color="item.status ? 'success' : 'error'" small dark>
          {{ item.status ? '启用' : '禁用' }}
        </v-chip>
      </template>
      <template v-slot:item.actions="{ item }">
        <v-icon small class="mr-2" @click="openDialog(item)">mdi-pencil</v-icon>
        <v-icon small color="error"         @click="remove(item)">mdi-delete</v-icon>
      </template>
    </v-data-table>

    <!-- 弹窗 -->
    <v-dialog v-model="dialogVisible" max-width="440px" persistent>
      <v-card>
        <v-card-title>{{ isEdit ? '编辑用户' : '新建用户' }}</v-card-title>
        <v-card-text>
          <v-form ref="form" v-model="valid">
            <v-text-field v-model="form.name"  label="姓名" :rules="[v => !!v || '必填']" outlined dense />
            <v-text-field v-model="form.email" label="邮箱" :rules="emailRules"            outlined dense />
          </v-form>
        </v-card-text>
        <v-card-actions>
          <v-spacer />
          <v-btn text @click="close">取消</v-btn>
          <v-btn color="primary" :loading="saving" @click="save">保存</v-btn>
        </v-card-actions>
      </v-card>
    </v-dialog>

    <!-- Snackbar 提示 -->
    <v-snackbar v-model="snack" :color="snackColor" top :timeout="3000">
      {{ snackText }}
      <template #action><v-btn text @click="snack=false">关闭</v-btn></template>
    </v-snackbar>
  </v-container>
</v-app>
