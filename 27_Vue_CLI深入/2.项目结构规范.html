4. public/ vs assets/ 的区别

public/ 目录

❌ 不经过 Webpack 处理

✅ 直接复制到 dist/

✅ 用绝对路径引用：/logo.png

适合： favicon、第三方不支持模块化的 JS、已优化的图片

src/assets/ 目录

✅ 经过 Webpack 处理

✅ 自动哈希文件名（缓存更新）

✅ 小图片自动转 Base64

适合： 组件用到的图片、字体、图标

<!-- public/ 中的资源：用绝对路径（.env 中的 VUE_APP_BASE_URL 可辅助）-->
<img src="/logo.png">
<link rel="icon" href="/favicon.ico">

<!-- assets/ 中的资源：用相对路径或 @ 别名（Webpack 处理）-->
<img src="@/assets/images/banner.png">
/* CSS 中 */
/* background: url(@/assets/images/bg.png) */

5. main.js 组织规范

main.js 是应用入口，职责是"注册和启动"，不应包含业务逻辑。

// src/main.js — 标准写法（分层清晰）
import Vue from 'vue'
import App from './App.vue'
import router from './router'
import store from './store'

// ① 全局样式
import './styles/index.scss'

// ② UI 组件库（按需引入更优）
import ElementUI from 'element-ui'
import 'element-ui/lib/theme-chalk/index.css'
Vue.use(ElementUI)

// ③ 全局配置
Vue.config.productionTip = false
Vue.config.devtools = process.env.NODE_ENV !== 'production'

// ④ 全局指令 / 过滤器（少量，多了用插件方式）
import { formatDate } from './utils/format'
Vue.filter('formatDate', formatDate)

// ⑤ 创建实例（最后执行）
new Vue({
  router,
  store,
  render: h => h(App)
}).$mount('#app')

6. 组件分层原则

合理的组件分层能让代码可复用、易维护。

三层架构

层级位置职责特征
页面层 views/ 与路由对应，协调数据流 dispatch action、传 props 给子组件、体积可以大
业务组件层 views/Xxx/ 子目录 页面专属的业务逻辑组件 可以访问 store，但只被一个页面使用
通用组件层 components/ 纯展示 / 通用交互，无业务逻辑 只通过 props/emit 通信，不访问 store，高度可复用
常见错误： 把 store.dispatch 写在通用组件里。通用组件应该只接收 props、触发 emit， 由父层（页面/业务组件）决定数据来源。这样组件才能真正跨项目复用。

7. API 请求层封装规范

// src/api/index.js — axios 实例
import axios from 'axios'
import store from '@/store'
import router from '@/router'

const http = axios.create({
  baseURL: process.env.VUE_APP_API_BASE_URL,  // 读环境变量
  timeout: 10000
})

// 请求拦截器：自动带上 token
http.interceptors.request.use(config => {
  const token = store.state.user.token
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

// 响应拦截器：统一错误处理
http.interceptors.response.use(
  res => res.data,
  err => {
    if (err.response?.status === 401) {
      store.dispatch('user/logout')
      router.push('/login')
    }
    return Promise.reject(err)
  }
)

export default http

// -----------------------------------------------
// src/api/user.js — 按模块拆分接口
import http from '.'

export const userApi = {
  login: (data) => http.post('/auth/login', data),
  logout: () => http.post('/auth/logout'),
  getProfile: () => http.get('/user/profile'),
  updateProfile: (data) => http.put('/user/profile', data)
}
