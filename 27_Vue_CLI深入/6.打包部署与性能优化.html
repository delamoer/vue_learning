1. npm run build 原理与产物

# 生产构建
npm run build

# 构建完成后，dist/ 目录结构：
dist/
├── index.html               ← 入口 HTML
├── favicon.ico
└── static/
    ├── js/
    │   ├── app.xxx.js       ← 应用代码（含路由非懒加载部分）
    │   ├── chunk-vendors.xxx.js   ← 第三方依赖（vue、vuex等）
    │   └── chunk-xxx.js     ← 懒加载的路由组件（各自单独文件）
    └── css/
        └── app.xxx.css

构建时做了什么？

优化项说明
代码压缩JS 经 Terser 压缩，CSS 经 cssnano 压缩
Tree-shaking移除未使用的代码（需要 ES Module 语法）
文件哈希文件名含内容哈希（app.abc123.js），用于缓存控制
代码拆包第三方库单独打包（chunk-vendors），浏览器可长期缓存
source map默认关闭（productionSourceMap: false），可按需开启

3. 路由懒加载（最有效的优化）

将每个路由对应的组件拆成独立 chunk，访问该路由时才加载，大幅减少首屏体积。

// ❌ 不懒加载（所有页面打进一个 app.js）
import HomeView from '../views/HomeView.vue'
import DashboardView from '../views/DashboardView.vue'

// ✅ 路由懒加载（每个页面独立 chunk，按需加载）
const routes = [
  {
    path: '/',
    component: () => import('../views/HomeView.vue')
  },
  {
    path: '/dashboard',
    // webpackChunkName：同名 chunk 会合并打包
    component: () => import(
      /* webpackChunkName: "admin" */
      '../views/DashboardView.vue'
    )
  },
  {
    path: '/users',
    component: () => import(
      /* webpackChunkName: "admin" */  // 与 dashboard 同名 → 合并到一个包
      '../views/UsersView.vue'
    )
  }
]

4. CDN 外链（减少打包体积）

将 Vue、Vue Router、Vuex、axios 等大型依赖从 Webpack 打包中排除，改为从 CDN 加载。

// vue.config.js
const isProd = process.env.NODE_ENV === 'production'

// CDN 配置表
const cdn = {
  // externals：告诉 Webpack "这些包不要打进去，从全局变量取"
  externals: {
    'vue': 'Vue',
    'vue-router': 'VueRouter',
    'vuex': 'Vuex',
    'axios': 'axios'
  },
  // 需要在 HTML 中手动引入的 CDN 链接
  js: [
    'https://cdn.jsdelivr.net/npm/vue@2.7.14/dist/vue.min.js',
    'https://cdn.jsdelivr.net/npm/vue-router@3.6.5/dist/vue-router.min.js',
    'https://cdn.jsdelivr.net/npm/vuex@3.6.2/dist/vuex.min.js',
    'https://cdn.jsdelivr.net/npm/axios@1.6.0/dist/axios.min.js'
  ]
}

module.exports = defineConfig({
  chainWebpack(config) {
    if (isProd) {
      // 生产环境：排除这些依赖，用 CDN
      config.externals(cdn.externals)

      // 在 HTML 模板中注入 CDN 链接
      config.plugin('html').tap(args => {
        args[0].cdn = cdn
        return args
      })
    }
  }
})
<!-- public/index.html — 配合 CDN 注入 -->
<head>
  <!-- <% if (htmlWebpackPlugin.options.cdn) { %> -->
  <% for (var js of htmlWebpackPlugin.options.cdn.js) { %>
  <script src="<%= js %>" crossorigin="anonymous"></script>
  <% } %>
  <!-- <% } %> -->
</head>

5. Gzip 压缩（构建时预压缩）

# 安装 compression-webpack-plugin
npm install -D compression-webpack-plugin
// vue.config.js
const CompressionPlugin = require('compression-webpack-plugin')

module.exports = defineConfig({
  chainWebpack(config) {
    if (process.env.NODE_ENV === 'production') {
      config.plugin('compression')
        .use(CompressionPlugin, [{
          algorithm: 'gzip',
          test: /\.(js|css|html|svg)$/,
          threshold: 10240,       // 大于 10KB 才压缩
          minRatio: 0.8           // 压缩率 > 80% 才生成
        }])
    }
  }
})
Nginx 配合： 生成了 .gz 文件后，需要 Nginx 开启 gzip_static on， 浏览器支持时直接返回 .gz 文件，省去动态压缩开销。

6. splitChunks 代码分割策略

// vue.config.js — 自定义拆包策略
module.exports = defineConfig({
  configureWebpack: {
    optimization: {
      splitChunks: {
        chunks: 'all',
        cacheGroups: {
          // 把 vue 全家桶单独打一个包（变化不频繁，可长期缓存）
          vue: {
            name: 'chunk-vue',
            test: /[\\/]node_modules[\\/](vue|vue-router|vuex)[\\/]/,
            priority: 20
          },
          // 把 UI 组件库单独打一个包
          elementUI: {
            name: 'chunk-elementUI',
            test: /[\\/]node_modules[\\/]element-ui[\\/]/,
            priority: 15
          },
          // 其余 node_modules 打进 vendors
          vendors: {
            name: 'chunk-vendors',
            test: /[\\/]node_modules[\\/]/,
            priority: 10
          }
        }
      }
    }
  }
})

7. 图片优化

// vue.config.js — 调整图片转 Base64 的阈值
module.exports = defineConfig({
  chainWebpack(config) {
    // 默认 < 4KB 的图片转 Base64，可根据需要调整
    config.module
      .rule('images')
      .use('url-loader')
      .tap(options => ({
        ...options,
        limit: 8192    // 改为 8KB
      }))
  }
})
