1安装与引入
完整引入（推荐入门）
// main.js
import Vue from 'vue'
import ElementUI from 'element-ui'
import 'element-ui/lib/theme-chalk/index.css'

Vue.use(ElementUI, { size: 'small', zIndex: 3000 })
// size: 'medium' | 'small' | 'mini'  （全局默认尺寸）
按需引入（生产推荐，减小打包体积）
// babel.config.js 插件配置
plugins: [[ 'component', { libraryName: 'element-ui', styleLibraryName: 'theme-chalk' }]]

// main.js 只引入需要的组件
import { Button, Table, Form, FormItem, Input, Select } from 'element-ui'
[Button, Table, Form, FormItem, Input, Select].forEach(c => Vue.component(c.name, c))
综合项目中的配置（vue2-admin-demo）：完整引入 + size: 'small' 全局 small 尺寸， 保持了代码简洁与原型演示速度，生产项目建议按需引入。

4Form 表单（重点）
三要素： ① :model 绑定数据对象 · ② :rules 配置校验规则 · ③ prop 指定字段名（用于触发校验） 
<!-- 模板 -->
<el-form ref="form" :model="form" :rules="rules" label-width="80px">
  <el-form-item label="用户名" prop="username">
    <el-input v-model="form.username" />
  </el-form-item>
  <el-form-item>
    <el-button type="primary" @click="submit">提交</el-button>
  </el-form-item>
</el-form>

// JS
data() {
  return {
    form: { username: '', email: '' },
    rules: {
      username: [
        { required: true, message: '请输入用户名', trigger: 'blur' },
        { min: 3, max: 20, message: '长度 3-20 位', trigger: 'blur' }
      ],
      email: [
        { required: true, message: '请输入邮箱' },
        { type: 'email', message: '邮箱格式不正确', trigger: 'blur' }
      ]
    }
  }
},
methods: {
  submit() {
    this.$refs.form.validate(valid => {
      if (valid) { /* 提交请求 */ }
    })
  },
  reset() { this.$refs.form.resetFields() }
}
方法（通过 ref 调用）说明
validate(callback)校验全部字段，回调参数为是否通过
validateField(prop, cb)校验单个字段
resetFields()重置所有字段为初始值并清除校验
clearValidate(props?)清除校验提示（不改变值）
5Table 表格（重点）
<!-- 基础 + 自定义列 + 操作列 -->
<el-table :data="tableData" stripe border style="width:100%">
  <!-- 普通列 -->
  <el-table-column prop="name"   label="姓名"  width="120" />
  <el-table-column prop="email"  label="邮箱" />

  <!-- 自定义渲染（作用域插槽） -->
  <el-table-column label="状态" width="90">
    <template v-slot="{ row }">
      <el-tag :type="row.status === 1 ? 'success' : 'danger'">
        {{ row.status === 1 ? '启用' : '禁用' }}
      </el-tag>
    </template>
  </el-table-column>

  <!-- 操作列 -->
  <el-table-column label="操作" width="150" fixed="right">
    <template v-slot="{ row }">
      <el-button size="mini" @click="edit(row)">编辑</el-button>
      <el-button size="mini" type="danger" @click="del(row)">删除</el-button>
    </template>
  </el-table-column>
</el-table>
⚠️ 自定义列必须用作用域插槽：通过 v-slot="{ row }" 拿到当前行数据， row 是整行对象，$index 是行下标，column 是列配置对象。
6Dialog 弹窗
<!-- :visible.sync 等价于 :visible + @update:visible -->
<el-dialog
  title="编辑用户"
  :visible.sync="dialogVisible"
  width="480px"
  :close-on-click-modal="false"   <-- 点遮罩不关闭（防误操作）
  @closed="resetForm"              <-- 弹窗完全关闭后重置表单
>
  <!-- 内容 -->
  <el-form ...> ... </el-form>

  <!-- 底部按钮（具名插槽） -->
  <template #footer>
    <el-button @click="dialogVisible = false">取消</el-button>
    <el-button type="primary" :loading="submitting" @click="confirm">确认</el-button>
  </template>
</el-dialog>
常见坑：Dialog 内的 Form 校验状态和 Dialog 生命周期解耦。 应在 @closed（完全关闭动画结束后）再 resetFields()， 否则 @close 时动画未结束会看到残留校验高亮。
7Pagination 分页
<el-pagination
  background
  layout="total, sizes, prev, pager, next, jumper"
  :total="total"        <-- 总条数
  :page-size="pageSize"
  :current-page="page"
  :page-sizes="[10, 20, 50, 100]"
  @current-change="onPageChange"   <-- 页码改变
  @size-change="onSizeChange"     <-- 每页条数改变
/>

// layout 字段说明：
// total  → 显示"共 xx 条"
// sizes  → 每页条数选择器
// prev   → 上一页
// pager  → 页码列表
// next   → 下一页
// jumper → 跳转输入框

9Switch 开关
<el-switch
  v-model="status"
  :active-value="1"           <-- 开启时绑定值（默认 true）
  :inactive-value="0"        <-- 关闭时绑定值（默认 false）
  active-text="启用"
  inactive-text="禁用"
  active-color="#67C23A"
  @change="onStatusChange"
/>

// 实战中常见写法：列表中切换状态后立即调接口
async onStatusChange(val, row) {
  try {
    await updateUserStatus(row.id, val)
    this.$message.success('状态更新成功')
  } catch {
    row.status = val === 1 ? 0 : 1  // 接口失败：回滚到原值
  }
}

  clearable                       <-- 显示清空按钮
  prefix-icon="el-icon-search"   <-- 前缀图标

/>

<el-input type="password" show-password />

<!-- Select 下拉框 -->
<el-select v-model="role" placeholder="选择角色" clearable>
  <el-option
    v-for="item in roleOptions"
    :key="item.value"
    :label="item.label"
    :value="item.value"
  />
</el-select>

// roleOptions 数据格式：
[
  { label: '管理员', value: 'admin' },
  { label: '编辑',   value: 'editor' }
]
⚠️ Select 绑定值类型：v-model 绑定的是 el-option 的 :value， 注意值类型要和后端保持一致（字符串 vs 数字），否则回显时会找不到匹配项显示空白。

12Message 消息 / MessageBox 弹框 / Notification 通知
// ① Message（顶部轻提示，2-3秒自动消失）
this.$message.success('操作成功')
this.$message.error('请求失败')
this.$message.warning('请注意')
this.$message({ message: '自定义', type: 'info', duration: 5000, showClose: true })

// ② MessageBox（确认弹框，async/await 用法）
async deleteUser(id) {
  try {
    await this.$confirm('确定删除该用户吗？', '提示', {
      confirmButtonText: '确定',
      cancelButtonText:  '取消',
      type: 'warning'
    })
    await deleteUserApi(id)
    this.$message.success('删除成功')
  } catch {
    // 用户点"取消" → catch 但不需要处理
  }
}

// ③ Notification（右上角通知，适合后台异步任务完成提醒）
this.$notify({
  title: '导出完成',
  message: '报表已生成，点击下载',
  type: 'success',
  duration: 0,        // 0 = 不自动关闭

})
